/// The error type every generated SDK method returns.
#[derive(Debug, thiserror::Error)]
pub enum Error {
    /// A non-zero CLI exit (sdk.errors.nonZeroExit).
    #[error("cli exited with code {exit_code}{}", stderr_suffix(.stderr))]
    Cli {
        exit_code: i32,
        stdout: String,
        stderr: String,
    },
    /// The invocation outlived the configured timeout and the child was killed.
    #[error("cli timed out")]
    Timeout,
    /// The CLI binary could not be spawned or awaited.
    #[error("{0}")]
    Spawn(String),
    /// A required argument was empty or invalid before the CLI was spawned.
    #[error("invalid argument: {0}")]
    InvalidArgument(String),
    /// A JSON (de)serialization error.
    #[error("decode error: {0}")]
    Decode(#[from] serde_json::Error),
}

/// `": <stderr>"` when stderr is non-empty — the Display suffix for `Error::Cli`.
fn stderr_suffix(stderr: &str) -> String {
    if stderr.is_empty() {
        String::new()
    } else {
        format!(": {stderr}")
    }
}
