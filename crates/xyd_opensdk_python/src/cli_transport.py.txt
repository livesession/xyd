from __future__ import annotations

import json
import os
import subprocess
from enum import Enum
from typing import Any, Optional

DEFAULT_BIN = __XYD_BIN__
"""The binary invoked when neither an explicit bin_path nor the BIN_ENV_VAR override is set; resolved via PATH."""

BIN_ENV_VAR = __XYD_BIN_ENV_VAR__
"""The environment variable checked for a binary path before falling back to DEFAULT_BIN on PATH."""

DEFAULT_TIMEOUT_MS = __XYD_TIMEOUT_MS__
"""Bounds one CLI invocation (sdk.timeout.defaultTimeoutMs, milliseconds); on expiry the child process is killed and the call fails with a timed-out CliError."""


class CommandResult:
    """The result of one successful CLI invocation. json() parses stdout as JSON."""

    def __init__(self, stdout: str, stderr: str, exit_code: int) -> None:
        self.stdout = stdout
        self.stderr = stderr
        self.exit_code = exit_code

    def json(self) -> Any:
        """Decode the invocation's stdout as JSON."""
        return json.loads(self.stdout)


class CliError(Exception):
    """A failed CLI invocation: a non-zero exit, or a timeout (exit_code -1, timed_out True)."""

    def __init__(self, exit_code: int, stdout: str, stderr: str, timed_out: bool = False) -> None:
        self.exit_code = exit_code
        self.stdout = stdout
        self.stderr = stderr
        self.timed_out = timed_out
        if timed_out:
            message = "cli timed out"
        else:
            message = "cli exited with code " + str(exit_code)
            if stderr:
                message += ": " + stderr
        super().__init__(message)


def _decode(data: Any) -> str:
    if data is None:
        return ""
    if isinstance(data, str):
        return data
    return data.decode("utf-8", "replace")


def cli_text(value: Any) -> str:
    """One argv token for a valued flag or positional (bools never reach here as bare-flag values)."""
    if isinstance(value, bool):
        return "true" if value else "false"
    if isinstance(value, Enum):
        # Enums stringify to their wire value, never the member name.
        return str(value.value)
    return str(value)


def cli_json(value: Any) -> str:
    """One compact-JSON argv token for a json-encoded flag value."""
    return json.dumps(value, separators=(",", ":"))


class Transport:
    """Spawns the CLI binary for every generated SDK method, captures
    stdout/stderr, enforces the configured timeout, and maps non-zero exits to
    CliError (sdk.errors.nonZeroExit)."""

    def __init__(
        self,
        bin_path: Optional[str] = None,
        env: Optional[dict[str, str]] = None,
        cwd: Optional[str] = None,
        timeout: Optional[float] = None,
    ) -> None:
        """bin_path overrides the binary (highest precedence); env adds
        variables on top of the inherited process environment; cwd sets the
        child's working directory; timeout is in MILLISECONDS
        (sdk.timeout.defaultTimeoutMs) — None keeps the default, 0 disables
        the deadline."""
        self.bin_path = bin_path
        self.env = env
        self.cwd = cwd
        self.timeout_ms = timeout if timeout is not None else DEFAULT_TIMEOUT_MS

    def run(self, argv: list[str]) -> CommandResult:
        """Execute the CLI with the given argv. Binary resolution precedence:
        explicit bin_path > the BIN_ENV_VAR environment variable > DEFAULT_BIN
        on PATH. Stdin is not attached (sdk.process.stdin)."""
        bin_ = self.bin_path or os.environ.get(BIN_ENV_VAR) or DEFAULT_BIN
        env = None
        if self.env:
            env = dict(os.environ)
            env.update(self.env)
        timeout_s = self.timeout_ms / 1000.0 if self.timeout_ms and self.timeout_ms > 0 else None
        try:
            completed = subprocess.run(
                [bin_] + argv,
                stdin=subprocess.DEVNULL,
                stdout=subprocess.PIPE,
                stderr=subprocess.PIPE,
                env=env,
                cwd=self.cwd,
                timeout=timeout_s,
            )
        except subprocess.TimeoutExpired as expired:
            # subprocess.run kills the child on expiry before raising.
            raise CliError(
                -1,
                _decode(expired.stdout),
                _decode(expired.stderr),
                timed_out=True,
            ) from None
        stdout = _decode(completed.stdout)
        stderr = _decode(completed.stderr)
        if completed.returncode != 0:
            raise CliError(completed.returncode, stdout, stderr)
        return CommandResult(stdout, stderr, 0)
