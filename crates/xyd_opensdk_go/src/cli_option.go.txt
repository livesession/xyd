package option

import (
	"time"

	"__XYD_MODULE__/internal/runner"
)

// CommandOption mutates the shared CLI runner configuration.
type CommandOption = runner.CommandOption

// WithBinPath overrides the CLI binary path (highest precedence — above the
// runner's env-var override and the PATH lookup of the default binary).
func WithBinPath(path string) CommandOption {
	return func(cfg *runner.Config) error {
		cfg.BinPath = path
		return nil
	}
}

// WithEnv adds an environment variable on top of the inherited process
// environment for the child CLI process.
func WithEnv(key, value string) CommandOption {
	return func(cfg *runner.Config) error {
		if cfg.Env == nil {
			cfg.Env = map[string]string{}
		}
		cfg.Env[key] = value
		return nil
	}
}

// WithCwd sets the child CLI process's working directory.
func WithCwd(dir string) CommandOption {
	return func(cfg *runner.Config) error {
		cfg.Cwd = dir
		return nil
	}
}

// WithTimeout bounds one CLI invocation; on expiry the child is killed and
// the call fails with a timed-out *runner.CliError. Zero keeps the default.
func WithTimeout(timeout time.Duration) CommandOption {
	return func(cfg *runner.Config) error {
		cfg.Timeout = timeout
		return nil
	}
}
