import { execFile } from 'node:child_process';

/**
 * The binary invoked when neither an explicit `binPath` option nor the
 * `BIN_ENV_VAR` override is set; resolved via PATH.
 */
const DEFAULT_BIN = __XYD_BIN__;

/**
 * The environment variable checked for a binary path before falling back to
 * `DEFAULT_BIN` on PATH.
 */
const BIN_ENV_VAR = __XYD_BIN_ENV_VAR__;

/**
 * Bounds one CLI invocation (sdk.timeout.defaultTimeoutMs); on expiry the
 * child process is killed and the call rejects with a timed-out `CliError`.
 */
const DEFAULT_TIMEOUT_MS = __XYD_TIMEOUT_MS__;

/** stdout/stderr capture cap per invocation. */
const MAX_BUFFER = 64 * 1024 * 1024;

/** Per-client CLI runner configuration. */
export interface ClientOptions {
  /**
   * Overrides the CLI binary path (highest precedence — above the env-var
   * override and the PATH lookup of the default binary).
   */
  binPath?: string;
  /**
   * Environment variables added on top of the inherited process environment
   * for the child CLI process.
   */
  env?: Record<string, string>;
  /** The child CLI process's working directory (default: inherited). */
  cwd?: string;
  /** Bounds one CLI invocation in milliseconds; on expiry the child is killed. */
  timeoutMs?: number;
}

/** The result of one successful CLI invocation. */
export class CommandResult {
  constructor(
    /** The invocation's captured standard output. */
    readonly stdout: string,
    /** The invocation's captured standard error. */
    readonly stderr: string,
    /** The exit code (always 0 — non-zero exits reject with `CliError`). */
    readonly exitCode: number,
  ) {}

  /** Decode the invocation's stdout as JSON. */
  json<T>(): T {
    return JSON.parse(this.stdout) as T;
  }
}

/**
 * A failed CLI invocation: a non-zero exit (sdk.errors.nonZeroExit), or a
 * timeout (`exitCode` -1, `timedOut` true).
 */
export class CliError extends Error {
  constructor(
    readonly exitCode: number,
    readonly stdout: string,
    readonly stderr: string,
    readonly timedOut: boolean = false,
  ) {
    super(
      timedOut
        ? 'cli timed out'
        : 'cli exited with code ' + exitCode + (stderr ? ': ' + stderr : ''),
    );
    this.name = 'CliError';
  }
}

/** The base class the generated client extends: binary resolution + spawning. */
export class CLIClient {
  readonly _options: ClientOptions;

  constructor(options: ClientOptions = {}) {
    this._options = options;
  }

  /**
   * Execute the CLI with the given argv. Binary resolution precedence: the
   * explicit `binPath` option > the `BIN_ENV_VAR` environment variable >
   * `DEFAULT_BIN` on PATH. Stdin is not attached (sdk.process.stdin).
   */
  _run(argv: string[]): Promise<CommandResult> {
    const bin = this._options.binPath || process.env[BIN_ENV_VAR] || DEFAULT_BIN;
    const timeout = this._options.timeoutMs ?? DEFAULT_TIMEOUT_MS;
    const env = this._options.env ? { ...process.env, ...this._options.env } : process.env;
    return new Promise((resolve, reject) => {
      const child = execFile(
        bin,
        argv,
        { timeout, maxBuffer: MAX_BUFFER, cwd: this._options.cwd, env },
        (error, stdout, stderr) => {
          if (error) {
            if (error.killed || error.signal) {
              reject(new CliError(-1, stdout, stderr, true));
            } else if (typeof error.code === 'number') {
              reject(new CliError(error.code, stdout, stderr));
            } else {
              reject(error);
            }
            return;
          }
          resolve(new CommandResult(stdout, stderr, 0));
        },
      );
      child.stdin?.end();
    });
  }
}

/** The base class every generated resource extends: it holds the client. */
export abstract class CLIResource {
  protected _client: CLIClient;

  constructor(client: CLIClient) {
    this._client = client;
  }
}
